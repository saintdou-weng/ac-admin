function budgetSheet_(){return sh_('BudgetDoc',['rev','index','json']);}
function budgetDocRead_(){
 const sh=budgetSheet_(),n=sh.getLastRow();if(n<2)return {rev:0,doc:null};
 const v=sh.getRange(2,1,n-1,3).getValues();let rev=0,parts=[];
 v.forEach(function(r){if(Number(r[0])>rev){rev=Number(r[0]);parts=[];}if(Number(r[0])===rev)parts[Number(r[1])]=String(r[2]).replace(/^j:/,'');});
 if(!rev)return {rev:0,doc:null};try{return {rev:rev,doc:JSON.parse(parts.join(''))};}catch(e){return {rev:rev,doc:null,error:'BAD_DOC'};}
}
function budgetDocWrite_(doc,rev){
 const sh=budgetSheet_(),s=JSON.stringify(doc),rows=[];
 for(let i=0,k=0;i<s.length;i+=AC_BUDGET.CELL,k++)rows.push([rev,k,'j:'+s.slice(i,i+AC_BUDGET.CELL)]);
 const old=budgetDocRead_();
 if(old.doc){const bk=sh_('BudgetBackup',['time','rev','who','json']);const os=JSON.stringify(old.doc);bk.appendRow([new Date(),old.rev,String(doc._who||''),os.length<49000?'j:'+os:'(too big) '+os.length]);
  const bn=bk.getLastRow();if(bn>41)bk.deleteRows(2,bn-41);}
 sh.getRange(2,1,rows.length,3).setNumberFormat('@').setValues(rows);
 const last=sh.getLastRow();if(last>rows.length+1)sh.getRange(rows.length+2,1,last-rows.length-1,3).clearContent();
}
function budgetItems_(d){if(!d)return 0;const c=function(o){return o?Object.keys(o).length:0;};return c(d.years)+c(d.notes)+c(d.actual)+c(d.cats)+(d.custom||[]).length;}
function budgetPut_(b){
 if(String(b.key||'')!==AC_BUDGET.KEY)return {ok:false,error:'BAD_KEY'};
 const doc=b.doc;if(!doc||typeof doc!=='object')return {ok:false,error:'NO_DOC'};
 const lock=LockService.getScriptLock();lock.waitLock(20000);
 try{
  const cur=budgetDocRead_();
  if(Number(b.baseRev||0)!==cur.rev)return {ok:true,conflict:true,rev:cur.rev,doc:cur.doc};
  const was=budgetItems_(cur.doc),now=budgetItems_(doc);
  if(was>=6&&now<was/2&&!b.force)return {ok:false,error:'SHRINK',msg:'新資料比雲端少一半以上，沒有覆蓋（'+now+' < '+was+'）'};
  doc._who=String(b.who||'').slice(0,40);doc._at=new Date().toISOString();
  const rev=cur.rev+1;budgetDocWrite_(doc,rev);
  return {ok:true,rev:rev};
 }finally{lock.releaseLock();}
}

}
function budgetCls_(d){d=String(d||'').toLowerCase().replace(/[\u200b\s]+/g,' ').trim();

function budgetPayLive_(fromYm,toYm){
 const p=PropertiesService.getScriptProperties(),man=budgetHp_('action=smartManifest&tool=hrpay_payroll'),hs=man.hashes||{},by={},out={};
 Object.keys(hs).forEach(function(k){const m=k.match(/^m:(\d{4}-\d{2}):p\d/);if(m&&m[1]>=fromYm&&m[1]<=toYm)(by[m[1]]=by[m[1]]||[]).push(k);});
 Object.keys(by).sort().forEach(function(ym){
  const sig=by[ym].sort().map(function(k){return k+'='+hs[k];}).join('|'),ck='BUD_PAY_'+ym,old=p.getProperty(ck);
  if(old){try{const o=JSON.parse(old);if(o.sig===Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5,sig))){out[ym]=o.a;return;}}catch(e){}}
  const a=[0,0,0,0,0,0],ix={DL:0,IDL:2,OFF:4};
  by[ym].forEach(function(k){const d=budgetHp_('action=smartBucket&tool=hrpay_payroll&bucket='+encodeURIComponent(k));(d.records||[]).forEach(function(rec){const r=rec.row;if(!r||r.id==null)return;
   let AG=budgetXl_(r,'AG');if(AG==null)AG=Number(r.subtotal)||0;let S=budgetXl_(r,'S');if(S==null)S=Number(r.advGiven)||0;const i=ix[budgetCls_(r.dept)];a[i]++;a[i+1]+=AG+S;});});
  a[1]=Math.round(a[1]);a[3]=Math.round(a[3]);a[5]=Math.round(a[5]);out[ym]=a;
  p.setProperty(ck,JSON.stringify({sig:Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5,sig)),a:a}));

}

function budgetTick_(){
