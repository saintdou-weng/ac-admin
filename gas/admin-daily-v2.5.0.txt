 const today=dailyToday_(),p=PropertiesService.getScriptProperties(),L=[],errors=[],keys={};let news=0;
 /* 人事：只用一句帶過，明細在 HRA 群組 */
 try{const a=dailyAtt_(),last=p.getProperty('DAILY_ATT_DATE')||'';
  if(a.date>last){const t=weeklyPlain_(a.html),rate=dailyNum_(t,/出勤率\s*([\d.]+)%/),due=dailyNum_(t,/(\d+)\s*\S*\s*應到/),att=dailyNum_(t,/(\d+)\s*\S*\s*出勤\s*\n/),np=dailyNum_(t,/(\d+)\s*\S*\s*NP/),jn=dailyNum_(t,/入職?\s*\+(\d+)/),rs=dailyNum_(t,/離職?\s*[−-](\d+)/);
   L.push('👥 <b>人事</b>　'+dailyMdw_(a.date)+' 出勤率 '+(rate||'—')+'%'+(att&&due?'（'+att+'/'+due+'）':'')+(np&&np!=='0'?'，無故缺勤 '+np+' 人':'')+((jn&&jn!=='0')||(rs&&rs!=='0')?'，入 '+(jn||0)+' 離 '+(rs||0):'')+'；明細已在 HRA 群組。');
   keys.att=a.date;news++;}}catch(e){errors.push('考勤：'+String(e&&e.message||e).slice(0,160));}
 /* 生產：一句重點，明細在 PSJERP */
 try{const d=dailyProdFetch_(),last=p.getProperty('DAILY_PROD_DATE')||'';
  if(d.date>last&&d.facts){const f=d.facts,s=f.sewing||{},w=f.wip||{},an=(f.anomalies||[]).slice().sort(function(a,b){return (b.sev||0)-(a.sev||0);})[0];
   let line='🏭 <b>生產</b>　'+dailyMdw_(d.date)+(s.rows?' 車縫 '+dailyN_(s.pieces)+' 件、'+dailyEffDot_(s.eff)+'效率 '+dailyP1_(s.eff)+'%':' 沒有車縫資料')+(w.daysOfOutput?'，已裁未縫約 '+dailyP1_(w.daysOfOutput)+' 天':'')+'；明細看 PSJERP。';
   if(an&&an.zh){let t=String(an.zh);if(t.length>50)t=t.slice(0,48)+'…';line+='\n　⚠️ '+esc_(t);}
   L.push(line);keys.prod=d.date;news++;}}catch(e){errors.push('生產：'+String(e&&e.message||e).slice(0,160));}
 /* 總務保全 */
 try{const g=dailyGaBrief_(p.getProperty('DAILY_SENT_AT'));if(g.fresh||g.stale){L.push(g.line);if(g.fresh)news++;}}catch(e){errors.push('總務保全：'+String(e&&e.message||e).slice(0,160));}
 /* ERP */
 try{const e=dailyErp_(),last=p.getProperty('DAILY_ERP_AT')||'';if(e&&e.at&&e.at>last&&e.at.slice(0,10)>=weeklyGaAddDays_(today,-2)){L.push('🗂 <b>ERP</b>　'+esc_(e.at.slice(5,16).replace('-','/'))+' 已更新'+(e.note?'：'+esc_(String(e.note).slice(0,60)):'')+'。');keys.erp=e.at;news++;}}catch(e){}
 const parts=news?[{id:'brief',kind:'admin',html:'🏛️ <b>VRT 每日綜合摘要</b>　'+dailyMdw_(today)+'\n'+L.join('\n')}]:[];
 return {today:today,parts:parts,errors:errors,keys:keys};
}
function dailyPost_(html,kind){
 const chunks=typeof weeklyChunks_==='function'?weeklyChunks_(html):[String(html).slice(0,4000)];let last=null;
 chunks.forEach(function(t,i){last=tgRequest_('sendMessage',{chat_id:AC_ADMIN.CHAT_ID,text:t,parse_mode:'HTML',disable_web_page_preview:true,reply_markup:i===chunks.length-1?JSON.stringify(weeklyKeyboard_(kind)):undefined});});
 return last;
}
function dailySend_(b){
 const p=PropertiesService.getScriptProperties();let sent=0;
 b.parts.forEach(function(part){dailyPost_(part.html,part.kind);sent++;});
 p.setProperty('DAILY_SENT',b.today);   /* 沒新東西也算今天處理過，不再重試 */
 if(sent){p.setProperty('DAILY_SENT_AT',new Date().toISOString());if(b.keys.att)p.setProperty('DAILY_ATT_DATE',b.keys.att);if(b.keys.prod)p.setProperty('DAILY_PROD_DATE',b.keys.prod);if(b.keys.erp)p.setProperty('DAILY_ERP_AT',b.keys.erp);
  try{p.setProperty('DAILY_TEXT',weeklyPlain_(b.parts[0].html).slice(0,8500));}catch(e){}}
 if(b.errors.length)p.setProperty('DAILY_LAST_ERROR',b.today+' '+b.errors.join(' | ').slice(0,480));else p.deleteProperty('DAILY_LAST_ERROR');
 return sent;
}
/* 每分鐘由 pollAdminTelegram 呼叫：週日、國定假日不發；一天最多一則；不再補發新消息 */
function dailyTick_(){
 try{
  const now=new Date(),tz=AC_ADMIN.TZ;if(Number(Utilities.formatDate(now,tz,'u'))===7)return;
  const hm=Utilities.formatDate(now,tz,'HH:mm'),today=dailyToday_(),p=PropertiesService.getScriptProperties();
  if(hm<AC_DAILY.AT||hm>AC_DAILY.UNTIL||p.getProperty('DAILY_SENT')===today)return;
  if(dailyHoliday_(today)){p.setProperty('DAILY_SENT',today);return;}
  const tries=String(p.getProperty('DAILY_TRY')||'').split('|'),n=tries[0]===today?Number(tries[1])||0:0;if(n>=3)return;
  p.setProperty('DAILY_TRY',today+'|'+(n+1));
  dailySend_(dailyBuild_());
 }catch(e){try{PropertiesService.getScriptProperties().setProperty('DAILY_LAST_ERROR',String(e&&e.message||e).slice(0,500));}catch(x){}}
}
