function weeklyMonday_(d){
 const ymd=Utilities.formatDate(d||new Date(),AC_ADMIN.TZ,'yyyy-MM-dd'),t=new Date(ymd+'T00:00:00Z'),dow=(t.getUTCDay()+6)%7;
 t.setUTCDate(t.getUTCDate()-dow);return t.toISOString().slice(0,10);
}
function weeklyMd_(ymd){return String(ymd||'').slice(5).replace(/^0/,'').replace('-0','/').replace('-','/');}
function weeklyPlain_(html){return String(html||'').replace(/<[^>]+>/g,'').replace(/&lt;/g,'<').replace(/&gt;/g,'>').replace(/&amp;/g,'&');}
function weeklyFetchProd_(fresh){
 const r=UrlFetchApp.fetch(AC_WEEKLY.PROD_URL,{method:'post',contentType:'text/plain;charset=utf-8',payload:JSON.stringify({action:'dailyReportWeekNow',fresh:fresh?1:0}),muteHttpExceptions:true,followRedirects:true});
 if(r.getResponseCode()!==200)throw new Error('生產平台 HTTP '+r.getResponseCode());
 const j=JSON.parse(r.getContentText()),d=j&&j.data&&j.data.html?j.data:j;
 if(!d||!d.ok||!d.html)throw new Error('生產平台沒有回傳週報');
 return d;
}

function weeklyFetchPart_(url){

 const j=JSON.parse(r.getContentText()),d=j&&j.data&&(j.data.text||j.data.html)?j.data:j;
 if(!d||d.ok===false||!(d.text||d.html))throw new Error(String(d&&d.error||'沒有回傳摘要'));return d;
}

/* 組出本週要發的訊息（不發）。回傳 {monday, parts:[{id,html,kind}], errors:[]} */
function weeklyBuild_(fresh){
 const monday=weeklyMonday_(),parts=[],errors=[];
 try{const d=weeklyFetchProd_(fresh);parts.push({id:'prod',kind:'prod',html:weeklyProdMessage_(d,monday),facts:d.facts||null,dataDate:d.dataDate});}catch(e){errors.push('生產：'+String(e&&e.message||e).slice(0,160));}
 const adm=[];
 [['hr','🧑‍💼 人事',AC_WEEKLY.HR_URL],['ga','🧹 總務保全',AC_WEEKLY.GA_URL]].forEach(function(s){
  if(!s[2])return;
  try{const d=weeklyFetchPart_(s[2]);adm.push(String(d.html||esc_(d.text)));}catch(e){errors.push(s[1]+'：'+String(e&&e.message||e).slice(0,160));}
 });
 if(adm.length){const today=Utilities.formatDate(new Date(),AC_ADMIN.TZ,'yyyy-MM-dd');parts.push({id:'admin',kind:'admin',html:'🧑‍💼 <b>人事行政週報｜HR & Admin</b>　'+weeklyMd_(monday)+'–'+weeklyMd_(today)+'\n'+adm.join('\n━━━━━━━━━━━━\n')});}
 return {monday:monday,parts:parts,errors:errors};
}
function weeklySend_(built){
 const p=PropertiesService.getScriptProperties();let sent=0;
 built.parts.forEach(function(part){
  const text=String(part.html).slice(0,4000);
  const res=tgRequest_('sendMessage',{chat_id:AC_ADMIN.CHAT_ID,text:text,parse_mode:'HTML',disable_web_page_preview:true,reply_markup:JSON.stringify(weeklyKeyboard_(part.kind))});
  sent++;
  try{p.setProperty('WEEKLY_TEXT_'+part.id,weeklyPlain_(text).slice(0,8500));if(part.facts)p.setProperty('WEEKLY_FACTS_'+part.id,JSON.stringify(part.facts).slice(0,8500));p.setProperty('WEEKLY_MSG_'+part.id,String(res&&res.result&&res.result.message_id||''));}catch(e){}
 });
 if(sent){p.setProperty('WEEKLY_SENT',built.monday);p.setProperty('WEEKLY_SENT_AT',new Date().toISOString());}
 if(built.errors.length)p.setProperty('WEEKLY_LAST_ERROR',built.errors.join(' | ').slice(0,500));else p.deleteProperty('WEEKLY_LAST_ERROR');
 return sent;
}
/* 每分鐘由 pollAdminTelegram 呼叫：週五 09:00 起、當週還沒發過才發；失敗最多再試 3 次 */
function weeklyTick_(){
 try{
  const now=new Date(),tz=AC_ADMIN.TZ;if(Number(Utilities.formatDate(now,tz,'u'))!==AC_WEEKLY.DOW)return;
  const hm=Utilities.formatDate(now,tz,'HH:mm');if(hm<AC_WEEKLY.AT||hm>AC_WEEKLY.UNTIL)return;
  const p=PropertiesService.getScriptProperties(),monday=weeklyMonday_(now);if(p.getProperty('WEEKLY_SENT')===monday)return;
  const tk='WEEKLY_TRY',tries=String(p.getProperty(tk)||'').split('|'),n=tries[0]===monday?Number(tries[1])||0:0;if(n>=3)return;
  p.setProperty(tk,monday+'|'+(n+1));
  const built=weeklyBuild_(false);if(!built.parts.length){p.setProperty('WEEKLY_LAST_ERROR',(built.errors.join(' | ')||'no parts').slice(0,500));return;}
  weeklySend_(built);
 }catch(e){try{PropertiesService.getScriptProperties().setProperty('WEEKLY_LAST_ERROR',String(e&&e.message||e).slice(0,500));}catch(x){}}
}

function weeklyClaude_(system,user){
 const p=PropertiesService.getScriptProperties(),key=p.getProperty('CLAUDE_API_KEY');if(!key)return null;
 const H={'x-api-key':key,'anthropic-version':'2023-06-01'};
 let model=p.getProperty('CLAUDE_MODEL');
 if(!model){try{const m=JSON.parse(UrlFetchApp.fetch('https://api.anthropic.com/v1/models?limit=50',{headers:H,muteHttpExceptions:true}).getContentText());const ids=(m.data||[]).map(function(x){return x.id;});model=ids.filter(function(x){return /sonnet/.test(x);})[0]||ids[0]||'';if(model)p.setProperty('CLAUDE_MODEL',model);}catch(e){}}
 if(!model)return null;
 const r=UrlFetchApp.fetch('https://api.anthropic.com/v1/messages',{method:'post',contentType:'application/json',headers:H,muteHttpExceptions:true,payload:JSON.stringify({model:model,max_tokens:700,system:system,messages:[{role:'user',content:user}]})});
 const j=JSON.parse(r.getContentText());if(r.getResponseCode()!==200)throw new Error(String(j&&j.error&&j.error.message||('HTTP '+r.getResponseCode())).slice(0,160));
 return String((j.content||[]).map(function(c){return c.text||'';}).join('')).trim();
}
function weeklyTokens_(q){
 const out=[],s=String(q||'').toLowerCase();
 (s.match(/[a-z]+\d*|\d+/g)||[]).forEach(function(w){if(w.length>=2||/\d/.test(w))out.push(w);});
 const zh=s.replace(/[^一-鿿]/g,' ').split(/\s+/);zh.forEach(function(seg){for(let i=0;i+2<=seg.length;i++)out.push(seg.slice(i,i+2));});
 const stop={'多少':1,'什麼':1,'怎麼':1,'為什':1,'這週':1,'本週':1,'今天':1,'一下':1,'可以':1,'是不':1,'不是':1,'如何':1,'請問':1,'幫我':1,'the':1,'is':1,'how':1,'what':1,'why':1,'many':1};
 return out.filter(function(w,i){return !stop[w]&&out.indexOf(w)===i;});
}
function weeklyLookup_(q){
 const p=PropertiesService.getScriptProperties(),tok=weeklyTokens_(q);if(!tok.length)return '';
 const hits=[];
 ['prod','admin'].forEach(function(id){
  const lines=String(p.getProperty('WEEKLY_TEXT_'+id)||'').split('\n');let section='';
  lines.forEach(function(line,i){
   if(/^[^\s　]/.test(line)&&!/^━/.test(line))section=line;
   const low=line.toLowerCase().replace(/l(\d+)/g,' l$1 ');let sc=0;tok.forEach(function(w){if(/^l?\d+$/.test(w)?(' '+low+' ').indexOf(' '+w+' ')>=0||low.indexOf(w+' ')>=0:low.indexOf(w)>=0)sc+=w.length;});
   if(sc&&line.trim()&&!/^━/.test(line))hits.push({sc:sc,i:i,id:id,line:line,section:section});
  });
 });
 if(!hits.length)return '';
 hits.sort(function(a,b){return b.sc-a.sc||a.i-b.i;});
 const top=hits.slice(0,5).sort(function(a,b){return a.id===b.id?a.i-b.i:(a.id<b.id?1:-1);}),out=[];
 const all={prod:String(p.getProperty('WEEKLY_TEXT_prod')||'').split('\n'),admin:String(p.getProperty('WEEKLY_TEXT_admin')||'').split('\n')};
 const add=function(l){if(l&&out.indexOf(l)<0)out.push(l);};
 top.forEach(function(h){
  if(h.section&&h.section!==h.line)add(h.section);
  add(h.line);
  if(h.line===h.section){const L=all[h.id];for(let k=h.i+1;k<L.length&&k<=h.i+3&&/^[\s　]/.test(L[k]);k++)add(L[k]);}   /* 標題行命中：把底下縮排的細項一起帶出來 */
 });
 return out.join('\n');
}

 const q=raw.replace(/^[?？]\s*/,'').replace(new RegExp('@'+own,'ig'),' ').trim();if(!q)return {text:'🤖 問法：開頭打 ？ 再接問題，或回覆週報直接問。\n例：？L1 效率　？本月排單進度　？已裁未縫\nAsk with a leading ? or reply to the weekly report.',markup:weeklyKeyboard_('prod')};
